import pandas as pd import streamlit as st st.set_page_config( layout="wide", page_title="Conciliação Contábil & DRE Dinâmica" ) st.title("Painel de Conciliação e DRE em Tempo Real") @st.cache_data def carregar_dados_padrao(): return pd.DataFrame({ "id": [1, 2, 3, 4, 5, 6], "data": [ "2026-03-01", "2026-03-02", "2026-03-03", "2026-03-04", "2026-03-05", "2026-03-06", ], "descricao": [ "Cliente A - Prestação de Serviço", "Fornecedor de Insumos", "Aluguel do Escritório", "Sistema de Gestão (SaaS)", "Cliente B - Honorários", "Tarifa Bancária", ], "debito": [0.0, 1800.0, 2200.0, 350.0, 0.0, 80.0], "credito": [6500.0, 0.0, 0.0, 0.0, 4500.0, 0.0], "classificacao": [ "Receita", "Custo", "Despesa", "Despesa", "Receita", "Despesa", ], }) # Sidebar para upload de arquivos st.sidebar.header("📁 Importação de Dados") arquivo_upload = st.sidebar.file_uploader( "Envie o arquivo CSV ou Excel (Extrato ou Domínio)", type=["csv", "xlsx", "xls"], ) if "extrato" not in st.session_state: st.session_state.extrato = carregar_dados_padrao() if arquivo_upload is not None: try: if arquivo_upload.name.endswith(".csv"): df_novo = pd.read_csv(arquivo_upload) else: df_novo = pd.read_excel(arquivo_upload) if "id" not in df_novo.columns: df_novo.insert(0, "id", range(1, len(df_novo) + 1)) if "classificacao" not in df_novo.columns: df_novo["classificacao"] = "Despesa" st.session_state.extrato = df_novo st.sidebar.success("Arquivo importado com sucesso!") except Exception as e: st.sidebar.error(f"Erro ao processar o arquivo: {e}") df = st.session_state.extrato # Tratamento de colunas de Débito e Crédito if "debito" not in df.columns and "credito" not in df.columns: if "valor" in df.columns: df["debito"] = df["valor"].apply(lambda x: abs(x) if x < 0 else 0.0) df["credito"] = df["valor"].apply(lambda x: x if x > 0 else 0.0) else: df["debito"] = 0.0 df["credito"] = 0.0 # Painel Superior: Movimentação de Débito e Crédito total_credito = df["credito"].sum() total_debito = df["debito"].sum() saldo_liquido_mov = total_credito - total_debito col_m1, col_m2, col_m3 = st.columns(3) col_m1.metric("Total Entradas (Crédito)", f"R$ {total_credito:,.2f}") col_m2.metric("Total Saídas (Débito)", f"R$ {total_debito:,.2f}") col_m3.metric( "Saldo Líquido no Período", f"R$ {saldo_liquido_mov:,.2f}", delta=( "Positivo" if saldo_liquido_mov >= 0 else "Negativo (Saídas maiores)" ), ) st.markdown("---") # Layout de Duas Colunas: Esquerda (Extrato / Classificação) | Direita (DRE) col_tabela, col_dre = st.columns([1.6, 1], gap="large") with col_tabela: st.subheader("1. Painel de Movimentação e Classificação") st.markdown( "Classifique cada linha como **Receita**, **Custo**, **Despesa** ou" " **Ignorar** para alimentar a DRE em tempo real." ) edited_df = st.data_editor( df, column_config={ "id": None, "data": st.column_config.TextColumn("Data", width="small"), "descricao": st.column_config.TextColumn("Descrição", width="medium"), "debito": st.column_config.NumberColumn( "Débito (R$)", format="R$ %.2f" ), "credito": st.column_config.NumberColumn( "Crédito (R$)", format="R$ %.2f" ), "classificacao": st.column_config.SelectboxColumn( "Classificação", help="Defina a natureza contábil do lançamento", options=["Receita", "Custo", "Despesa", "Ignorar"], required=True, ), }, disabled=[ col for col in df.columns if col not in ["classificacao", "observacao"] ], hide_index=True, key="editor_movimentacao_completo", use_container_width=True, ) st.session_state.extrato = edited_df with col_dre: st.subheader("2. DRE Dinâmica") df_atual = st.session_state.extrato receita_bruta = df_atual[df_atual["classificacao"] == "Receita"][ "credito" ].sum() - df_atual[df_atual["classificacao"] == "Receita"]["debito"].sum() custos = df_atual[df_atual["classificacao"] == "Custo"]["debito"].sum() despesas = df_atual[df_atual["classificacao"] == "Despesa"]["debito"].sum() lucro_bruto = receita_bruta - custos resultado_liquido = lucro_bruto - despesas st.metric("Receita Bruta", f"R$ {receita_bruta:,.2f}") st.metric("(-) Custos Operacionais", f"R$ {custos:,.2f}") st.metric("(=) Lucro Bruto", f"R$ {lucro_bruto:,.2f}") st.metric("(-) Despesas Administrativas", f"R$ {despesas:,.2f}") cor_resultado = "normal" if resultado_liquido >= 0 else "inverse" st.metric( "(=) Resultado Líquido", f"R$ {resultado_liquido:,.2f}", delta=("Lucro" if resultado_liquido >= 0 else "Prejuízo"), delta_color=cor_resultado, ) st.markdown("---") st.markdown("##### Resumo Estruturado da DRE") dre_resumo = pd.DataFrame({ "Estrutura": [ "Receita Bruta", "(-) Custos", "(=) Lucro Bruto", "(-) Despesas", "(=) Resultado Líquido", ], "Total (R$)": [ receita_bruta, -custos, lucro_bruto, -despesas, resultado_liquido, ], }) st.dataframe(dre_resumo, hide_index=True, use_container_width=True)
